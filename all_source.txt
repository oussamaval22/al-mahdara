function PickerModal({ visible, items, selected, onSelect, onClose, title }: {visible:boolean;items:string[];selected:string;onSelect:(v:string)=>void;onClose:()=>void;title:string}) {
  return <Modal transparent visible={visible} animationType="slide" onRequestClose={onClose}><Pressable style={m.overlay} onPress={onClose}/><View style={m.sheet}><View style={m.handle}/><Text style={m.title}>{title}</Text><FlatList data={items} keyExtractor={x=>x} renderItem={({item})=><TouchableOpacity style={[m.item,item===selected&&m.selected]} onPress={()=>{onSelect(item);onClose();}}><Text style={[m.itemText,item===selected&&m.selectedText]}>{item}</Text>{item===selected&&<Ionicons name="checkmark" size={20} color={colors.green}/>}</TouchableOpacity>}/></View></Modal>;
}

export default function AudioScreen() {
  const [surahIdx,setSurahIdx]=useState(0); const [reciter,setReciter]=useState(''); const [rewaya,setRewaya]=useState(''); const [modal,setModal]=useState<'surah'|'reciter'|'rewaya'|null>(null);
  const surah=quran[surahIdx]; const audios=surah?.audio ?? [];
  const reciters=useMemo(()=>Array.from(new Set(audios.map(a=>a.reciter.ar))),[audios]);
  const rewayas=useMemo(()=>Array.from(new Set(audios.filter(a=>!reciter||a.reciter.ar===reciter).map(a=>a.rewaya.ar))),[audios,reciter]);
  const selected=useMemo<AudioItem|null>(()=>audios.find(a=>a.reciter.ar===reciter&&(!rewaya||a.rewaya.ar===rewaya)) ?? audios[0] ?? null,[audios,reciter,rewaya]);
  const player=useAudioPlayer(selected?.link ?? null,{updateInterval:500}); const status=useAudioPlayerStatus(player);

  const canUseNativeBackgroundAudio = Constants.executionEnvironment !== ExecutionEnvironment.StoreClient;

  useEffect(()=>{
    setAudioModeAsync({
      playsInSilentMode:true,
      shouldPlayInBackground:canUseNativeBackgroundAudio,
      interruptionMode:'doNotMix'
    }).catch(()=>{});
  },[canUseNativeBackgroundAudio]);
  useEffect(()=>{const first=audios[0]; if(first){setReciter(first.reciter.ar);setRewaya(first.rewaya.ar);}},[surahIdx]);
  useEffect(()=>{ if(reciter&&!rewayas.includes(rewaya)) setRewaya(rewayas[0] ?? ''); },[reciter,rewayas]);

  const play=()=>{
    if(!selected?.link){
      Alert.alert('تنبيه','لا يوجد رابط صوت لهذه التلاوة.');
      return;
    }
    // Expo Go لا يملك خدمة AudioControlsService الخاصة بالتطبيق المخصص،
    // لذلك لا نستدعي lock-screen controls داخله. في EAS/Standalone تعمل تلقائياً.
    if (canUseNativeBackgroundAudio) {
      try {
        player.setActiveForLockScreen?.(true,{
          title:surah?.name.ar ?? 'القرآن الكريم',
          artist:selected.reciter.ar,
          albumTitle:selected.rewaya.ar
        });
      } catch {}
    }
    player.play();
  };
  const format=(n:number)=>`${Math.floor(n/60)}:${Math.floor(n%60).toString().padStart(2,'0')}`;
  const progress=status.duration>0?Math.min(1,status.currentTime/status.duration):0;
  const next=()=>setSurahIdx(i=>Math.min(quran.length-1,i+1)); const prev=()=>setSurahIdx(i=>Math.max(0,i-1));
  const items=modal==='surah'?quran.map(s=>`${s.number}. ${s.name.ar}`):modal==='reciter'?reciters:rewayas;
  const selectedItem=modal==='surah'?`${surah?.number}. ${surah?.name.ar}`:modal==='reciter'?reciter:rewaya;
  return <SafeAreaView style={s.safe} edges={['bottom']}>
    <View style={s.content}>
      <View style={s.selectRow}><Select title="السورة" value={surah?.name.ar ?? '—'} onPress={()=>setModal('surah')}/><Select title="القارئ" value={reciter||'—'} onPress={()=>setModal('reciter')}/><Select title="الرواية" value={rewaya||'—'} onPress={()=>setModal('rewaya')}/></View>
      <LinearGradient colors={[colors.greenDark,colors.green]} start={{x:0,y:0}} end={{x:1,y:1}} style={s.player}>
        <View style={s.headphone}><Ionicons name="headset-outline" size={30} color={colors.gold}/></View><Text style={s.surah}>{surah?.name.ar}</Text><Text style={s.reciter}>{selected?.reciter.ar||'—'}</Text><Text style={s.rewaya}>{selected?.rewaya.ar||'—'}</Text>
        <View style={s.progress}><View style={s.track}><View style={[s.fill,{width:`${progress*100}%`}]}/></View><View style={s.times}><Text style={s.time}>{format(status.currentTime||0)}</Text><Text style={s.time}>{format(status.duration||0)}</Text></View></View>
        <View style={s.controls}><TouchableOpacity style={s.control} onPress={prev}><Ionicons name="play-skip-back" size={22} color={colors.white}/></TouchableOpacity><TouchableOpacity style={s.play} onPress={()=>status.playing?player.pause():play()}><Ionicons name={status.playing?'pause':'play'} size={30} color={colors.greenDark}/></TouchableOpacity><TouchableOpacity style={s.control} onPress={next}><Ionicons name="play-skip-forward" size={22} color={colors.white}/></TouchableOpacity></View>
      </LinearGradient>
      <View style={s.info}><Text style={s.infoText}>{surah?.revelation_place.ar}</Text><Text style={s.infoText}>{surah?.verses_count} آية</Text></View>
    </View>
    <PickerModal visible={modal!==null} items={items} selected={selectedItem} onSelect={v=>{if(modal==='surah'){const i=items.indexOf(v);if(i>=0)setSurahIdx(i);}else if(modal==='reciter'){setReciter(v)}else setRewaya(v)}} onClose={()=>setModal(null)} title={modal==='surah'?'اختر السورة':modal==='reciter'?'اختر القارئ':'اختر الرواية'}/>
  </SafeAreaView>;
}

import React, { useCallback, useState } from 'react';
import { View, Text, FlatList, TouchableOpacity, StyleSheet, Alert } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { useFocusEffect } from '@react-navigation/native';
import Ionicons from '@expo/vector-icons/Ionicons';
import { colors, shadow } from '../theme/colors';
import { Bookmark } from '../data/quranTypes';
import { quran } from '../data/quranData';
import { getBookmarks, removeBookmark } from '../storage/bookmarkStorage';
import { getPageBookmarks, removePageBookmark, PageBookmark } from '../storage/pageBookmarkStorage';

export default function BookmarksScreen({ navigation }: any) {
  const [surahs, setSurahs] = useState<Bookmark[]>([]);
  const [pages, setPages] = useState<PageBookmark[]>([]);

  const load = useCallback(async () => {
    const [savedSurahs, savedPages] = await Promise.all([getBookmarks(), getPageBookmarks()]);
    setSurahs(savedSurahs.sort((a, b) => b.savedAt - a.savedAt));
    setPages(savedPages.sort((a, b) => b.savedAt - a.savedAt));
  }, []);

  useFocusEffect(useCallback(() => { load(); }, [load]));

  const onRemoveSurah = (num: number, name: string) => {
    Alert.alert('حذف المحفوظة', `هل تريد حذف سورة "${name}" من المحفوظات؟`, [
      { text: 'إلغاء', style: 'cancel' },
      { text: 'حذف', style: 'destructive', onPress: async () => { await removeBookmark(num); load(); } },
    ]);
  };

  const onRemovePage = (page: number) => {
    Alert.alert('حذف الصفحة', `حذف الصفحة ${page} من المحفوظات؟`, [
      { text: 'إلغاء', style: 'cancel' },
      { text: 'حذف', style: 'destructive', onPress: async () => { await removePageBookmark(page); load(); } },
    ]);
  };

  const hasAny = pages.length > 0 || surahs.length > 0;

  if (!hasAny) {
    return (
      <SafeAreaView style={s.safe}>
        <View style={s.empty}>
          <View style={s.emptyIcon}><Ionicons name="bookmark-outline" size={46} color={colors.muted} /></View>
          <Text style={s.emptyTitle}>لا توجد محفوظات</Text>
          <Text style={s.emptySub}>من صفحة المصحف اضغط على الشاشة ثم اختر «حفظ الصفحة».</Text>
        </View>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={s.safe}>
      <FlatList
        data={pages}
        keyExtractor={(x) => `page-${x.page}`}
        contentContainerStyle={s.list}
        ListHeaderComponent={pages.length > 0 ? <Text style={s.sectionTitle}>الصفحات المحفوظة</Text> : null}
        renderItem={({ item }) => (
          <TouchableOpacity style={s.card} onPress={() => navigation.navigate('Reader', { pageNumber: item.page })} activeOpacity={0.8}>
            <View style={s.iconWrap}><Ionicons name="bookmark" size={20} color={colors.goldDark} /></View>
            <View style={s.info}><Text style={s.name}>صفحة {item.page}</Text><Text style={s.meta}>فتح الصفحة في المصحف</Text></View>
            <TouchableOpacity style={s.removeBtn} hitSlop={{ top: 10, bottom: 10, left: 10, right: 10 }} onPress={() => onRemovePage(item.page)}>
              <Ionicons name="trash-outline" size={19} color={colors.danger} />
            </TouchableOpacity>
          </TouchableOpacity>
        )}
        ListFooterComponent={surahs.length > 0 ? (
          <View>
            <Text style={s.sectionTitle}>السور المحفوظة</Text>
            {surahs.map((item) => (
              <TouchableOpacity key={item.surahNumber} style={s.card} onPress={() => navigation.navigate('Reader', { pageNumber: quran.find((s) => s.number === item.surahNumber)?.verses[0]?.page ?? 1 })} activeOpacity={0.8}>
                <View style={s.iconWrap}><Ionicons name="star" size={20} color={colors.gold} /></View>
                <View style={s.info}><Text style={s.name}>{item.surahName}</Text><Text style={s.meta}>سورة رقم {item.surahNumber}</Text></View>
                <TouchableOpacity style={s.removeBtn} hitSlop={{ top: 10, bottom: 10, left: 10, right: 10 }} onPress={() => onRemoveSurah(item.surahNumber, item.surahName)}>
                  <Ionicons name="trash-outline" size={19} color={colors.danger} />
                </TouchableOpacity>
              </TouchableOpacity>
            ))}
          </View>
        ) : null}
      />
    </SafeAreaView>
  );
}

export default function ReaderScreen({ route, navigation }: any) {
  const requested = Number(route.params?.pageNumber ?? 1);
  const initial = Math.min(PAGE_COUNT, Math.max(1, requested));
  const [page, setPage] = useState(initial);
  const [controlsVisible, setControlsVisible] = useState(false);
  const [saved, setSaved] = useState(false);
  const ref = useRef<FlatList<number>>(null);
  const data = useMemo(() => quranPages.map((p) => p.page), []);

  useEffect(() => {
    isPageBookmarked(initial).then(setSaved);
  }, [initial]);

  useEffect(() => {
    isPageBookmarked(page).then(setSaved);
  }, [page]);

  const jump = (target: number) => {
    const p = Math.min(PAGE_COUNT, Math.max(1, target));
    setPage(p);
    ref.current?.scrollToIndex({ index: p - 1, animated: false });
  };

  const toggleSave = async () => {
    const value = await togglePageBookmark(page);
    setSaved(value);
  };

  return (
    <SafeAreaView style={s.safe} edges={['bottom']}>
      <View style={s.topbar}>
        <TouchableOpacity onPress={() => jump(page - 1)} disabled={page === 1} style={s.nav}>
          <Ionicons name="chevron-back" size={22} color={page === 1 ? colors.border : colors.green} />
        </TouchableOpacity>
        <View style={s.pageBadge}>
          <Text style={s.pageText}>صفحة {page} / {PAGE_COUNT}</Text>
        </View>

          <Ionicons name="chevron-forward" size={22} color={page === PAGE_COUNT ? colors.border : colors.green} />
        </TouchableOpacity>
      </View>

      <FlatList
        ref={ref}
        data={data}
        horizontal
        pagingEnabled
        inverted

        showsHorizontalScrollIndicator={false}
        keyExtractor={(x) => String(x)}

import AsyncStorage from '@react-native-async-storage/async-storage';
import { Bookmark } from '../data/quranTypes';

const KEY = '@al_muhadhara_bookmarks';

export async function getBookmarks(): Promise<Bookmark[]> {
  const raw = await AsyncStorage.getItem(KEY);
  if (!raw) return [];
  try {
    const list = JSON.parse(raw) as Bookmark[];
    return Array.isArray(list) ? list : [];
  } catch {
    return [];
  }
}

export async function isBookmarked(surahNumber: number): Promise<boolean> {
  const list = await getBookmarks();
  return list.some((b) => b.surahNumber === surahNumber);
}

export async function toggleBookmark(surahNumber: number, surahName: string): Promise<boolean> {
  const list = await getBookmarks();
  const exists = list.some((b) => b.surahNumber === surahNumber);
  let next: Bookmark[];
  if (exists) {
    next = list.filter((b) => b.surahNumber !== surahNumber);
  } else {
    next = [{ surahNumber, surahName, savedAt: Date.now() }, ...list];
  }
  await AsyncStorage.setItem(KEY, JSON.stringify(next));
  return !exists;
}

export async function removeBookmark(surahNumber: number): Promise<void> {
  const list = await getBookmarks();
  const next = list.filter((b) => b.surahNumber !== surahNumber);
  await AsyncStorage.setItem(KEY, JSON.stringify(next));
}

}
